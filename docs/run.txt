# =============================================================================
# Hướng Dẫn Chạy Nhanh SwiftRide Backend
# =============================================================================

# --- CÁCH 1: CHẠY TRỌN GÓI BẰNG DOCKER COMPOSE (Khuyến nghị) ---
# 1. Chuẩn bị file .env:
cp .env.example .env

# 2. Khởi động toàn bộ cụm hệ thống (3 Services, Gateway, 2 Postgres, Mongo, RabbitMQ, Aspire):
docker compose up -d --build

# 3. Chạy smoke test kiểm thử tự động toàn bộ luồng (đặt xe -> thanh toán -> Paid):
bash scripts/smoke.sh

# 4. Truy cập giao diện:
# - API Gateway: http://localhost:5000
# - .NET Aspire Dashboard: http://localhost:18888
# - RabbitMQ Management: http://localhost:15672 (swiftride / swiftride123)

# 5. Dừng hệ thống:
docker compose down


# --- CÁCH 2: CHẠY PHÁT TRIỂN CỤC BỘ (dotnet run) ---
# 1. Khởi động các container cơ sở dữ liệu và message broker:
docker compose up -d postgres-trip postgres-payment mongo rabbitmq otel

# 2. Cập nhật migration cho cả 2 cơ sở dữ liệu PostgreSQL:
dotnet ef database update --project src/TripService/TripService.Infrastructure --startup-project src/TripService/TripService.Api
dotnet ef database update --project src/PaymentService/PaymentService.Infrastructure --startup-project src/PaymentService/PaymentService.Api

# 3. Chạy từng service trong các terminal riêng:
# - Terminal 1 (TripService): dotnet run --project src/TripService/TripService.Api
# - Terminal 2 (MatchingService): dotnet run --project src/MatchingService/MatchingService.Api
# - Terminal 3 (PaymentService): dotnet run --project src/PaymentService/PaymentService.Api
# - Terminal 4 (ApiGateway): dotnet run --project src/ApiGateway

# 4. Chạy toàn bộ 270 tests kiểm thử:
dotnet test SwiftRide.slnx
